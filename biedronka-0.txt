                expireDays = 1;
            } else if (perWeek) {
                let maxDays = 7;

                if (visibleEndedRaw) {
                    let visibleEnded = new Date(visibleEndedRaw.replace(" ", "T"));
                    let now = new Date();

                    // Difference in ms → conversion to days
                    let diffMs = visibleEnded.getTime() - now.getTime();

                    // The cookie cannot extend beyond the visibility of the popup
                    if (diffDays > 0 && diffDays < maxDays) {
                        expireDays = visibleEnded;
                    } else {
                        expireDays = maxDays;
                    }
                } else {
                    // none visible_ended → standard 7 days
                    expireDays = maxDays;
                }

            $.cookie(cookieName, countOfShowing,{
